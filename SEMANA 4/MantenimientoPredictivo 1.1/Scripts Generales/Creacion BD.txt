CREATE DATABASE IF NOT EXISTS mantenimiento_predictivo;
USE mantenimiento_predictivo;

-- Tabla de Usuarios (para el Login y Roles)
CREATE TABLE usuarios (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    password VARCHAR(255) NOT NULL,
    rol ENUM('administrador', 'mantenimiento', 'operario') NOT NULL,
    creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Tabla de Sensores (Iniciamos con 4 sensores)
CREATE TABLE sensores (
    id INT AUTO_INCREMENT PRIMARY KEY,
    codigo_sensor VARCHAR(50) UNIQUE NOT NULL,
    nombre_equipo VARCHAR(100) NOT NULL,
    tipo_variable VARCHAR(50) NOT NULL, -- Ej: Temperatura, Vibración, Presión
    estado ENUM('activo', 'inactivo', 'alerta') DEFAULT 'activo',
    actualizado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);
